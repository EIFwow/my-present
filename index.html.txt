        /* 1. 화면 전체를 180도 뒤집어버리는 마법 */
        body {
            display: flex;
            flex-direction: column;
            align-items: center;
            justify-content: center;
            height: 100vh;
            margin: 0;
            background-color: #f0f0f0;
            transform: rotate(180deg); /* 이 줄을 지우면 똑바로 돌아옵니다 */
            transition: transform 1s;
        }

        h1 { font-size: 2rem; color: #333; }

        /* 2. 도망가는 버튼 스타일 */
        #runaway-btn {
            padding: 15px 30px;
            font-size: 20px;
            background-color: #ff4757;
            color: white;
            border: none;
            border-radius: 10px;
            cursor: pointer;

        }

<body>

    <h1>야! 너를 위해 준비한 선물이 여기 있어!</h1>
    <p>아래 버튼을 클릭해서 선물을 확인해봐!</p>
    
    <button id="runaway-btn">🎁 선물 받기</button>

    <script>
        const btn = document.getElementById('runaway-btn');

        // 마우스가 버튼 근처에 가면 도망가게 만들기
        btn.addEventListener('mouseover', () => {
            const x = Math.random() * (window.innerWidth - btn.clientWidth);
            const y = Math.random() * (window.innerHeight - btn.clientHeight);
            
            btn.style.left = x + 'px';
            btn.style.top = y + 'px';
        });

        // 혹시라도 클릭에 성공했을 때의 반응
        btn.addEventListener('click', () => {
            alert('오... 이걸 클릭하네? 집요한 녀석;');
            alert('하지만 선물은 없단다. 메롱!');
        });
    </script>
